        <p id="open-wrap" style="display: none;">

    <script>
        var appType = 'happ';

        function buildIncyDeepLink(sub) {
            var value = (sub || '').trim();
            try {
                if (value.indexOf('%') !== -1) {
                    value = decodeURIComponent(value);
                }
            } catch (e) {}

            if (/^incy:\/\//i.test(value)) {
                return value;
            }
            return 'incy://import/' + value;
        }

        function buildHappDeepLink(urlParam) {
            var decoded = decodeURIComponent(urlParam);
            return /^happ:\/\//i.test(decoded) ? decoded : 'happ://add/' + decoded;
        }

        function buildDodoDeepLink(urlParam) {
            var value = (urlParam || '').trim();
            try {
                if (value.indexOf('%') !== -1) {
                    value = decodeURIComponent(value);
                }
            } catch (e) {}

            if (/^dodoapp:\/\//i.test(value)) {
                return value;
            }
            return 'dodoapp://import-sub?url=' + encodeURIComponent(value);
        }

        (function() {
            var params = new URLSearchParams(window.location.search);
            var dodo = params.get('dodo');
            var sub = params.get('sub');
            var urlParam = params.get('url_ha') || params.get('url') || params.get('q');
            var deepLink = null;

            // DoDo — отдельный параметр, проверяем первым
            if (dodo) {
                appType = 'dodo';
                deepLink = buildDodoDeepLink(dodo);
            } else if (sub) {
                appType = 'incy';
                deepLink = buildIncyDeepLink(sub);
            } else if (urlParam) {
                appType = 'happ';
                deepLink = buildHappDeepLink(urlParam);
            }

            if (!deepLink) {
                document.querySelector('.msg').textContent =
                    'Не указан параметр: ?dodo=... (DoDo), ?sub=... (Incy) или ?url_ha=... (Happ)';
                return;
            }

            /*
             * ТОЛЬКО меняем текст страницы.
             * Deep link ниже остаётся без изменений.
             */
            var appName;

            if (appType === 'dodo') {
                appName = 'DoDo';
            } else if (appType === 'incy') {
                appName = 'Incy';
            } else {
                appName = 'Happ';
            }

            document.getElementById('app-title').textContent =
                'Переход в ' + appName;

            document.getElementById('app-message').textContent =
                'Открываем ' + appName + '...';

            document.getElementById('open-app').textContent =
                'Открыть ' + appName;

            document.getElementById('fallback-message').textContent =
                'Если ' + appName + ' не открылось автоматически:';

            document.getElementById('store-button').textContent =
                'Скачать ' + appName;

            // ЭТО БЫЛО В ОРИГИНАЛЕ — НЕ МЕНЯЕМ
            document.getElementById('open-app').href = deepLink;
            document.getElementById('open-wrap').style.display = 'block';

            // ЭТО БЫЛО В ОРИГИНАЛЕ — НЕ МЕНЯЕМ
            window.location.replace(deepLink);

            setTimeout(function() {
                document.getElementById('fallback').style.display = 'block';
            }, 2000);
        })();

        function redirectToStore() {
            if (appType === 'dodo') {
                if (/iPhone|iPad|iPod|Macintosh/i.test(navigator.userAgent)) {
                    window.location.href = 'https://apps.apple.com/tj/app/dodo-app-proxy/id6759006686';
                } else {
                    window.location.href = 'https://apps.apple.com/tj/app/dodo-app-proxy/id6759006686';
                }
            } else if (appType === 'incy') {
                if (/iPhone|iPad|iPod|Macintosh/i.test(navigator.userAgent)) {
                    window.location.href = 'https://apps.apple.com/ru/app/incy/id6756943388';
                } else if (/Android/i.test(navigator.userAgent)) {
                    window.location.href = 'https://play.google.com/store/apps/details?id=llc.itdev.incy';
                } else {
                    window.location.href = 'https://incy.cc';
                }
            } else {
                if (/iPhone|iPad|iPod/i.test(navigator.userAgent)) {
                    window.location.href = 'https://apps.apple.com/ru/app/happ-proxy-utility-plus/id6746188973';
                } else if (/Android/i.test(navigator.userAgent)) {
                    window.location.href = 'https://play.google.com/store/apps/details?id=com.happproxy&hl=ru';
                } else {
                    window.location.href = 'https://github.com/Happ-proxy/happ-desktop/releases';
                }
            }
        }
    </script>
